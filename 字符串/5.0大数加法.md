## 5.0大数加法

题号： BM86
描述： 
思路： 
时间复杂度： $O(n)$
空间复杂度： $O(1)$

### 模拟法

就和 [链表相加](../链表/1.1链表相加.md) 一样

```python
def solve(self , s , t ):
    res = ""
    i, j, carry = len(s) - 1, len(t) - 1, 0
    while i >= 0 or j >= 0:
        n1 = int(s[i]) if i >= 0 else 0
        n2 = int(t[j]) if j >= 0 else 0
        tmp = n1 + n2 + carry
        carry = tmp // 10
        res = str(tmp % 10) + res  # 效率低，因为str不可变，需要复制
        i, j = i - 1, j - 1
    return "1" + res if carry else res
```



更优的方法： 使用列表append

```python
def AddLongInteger(s, t):
    res = []
    i, j, carry = len(s) - 1, len(t) - 1, 0

    while i >= 0 or j >= 0:
        n1 = int(s[i]) if i >= 0 else 0
        n2 = int(t[j]) if j >= 0 else 0

        tmp = n1 + n2 + carry
        carry = tmp // 10

        #res.append(str(tmp % 10))
        # 避免整数 → 字符串转换
        res.append(chr(tmp % 10 + 48))  # ord('0') = 48

        i, j = i - 1, j - 1

    if carry:
        res.append('1')

    return ''.join(reversed(res))
```

最优：

ord：字符 → 数字
chr：数字 → 字符

```python
def AddLongInteger(s: str, t: str) -> str:
    i = len(s) - 1
    j = len(t) - 1
    carry = 0
    res = []

    while i >= 0 or j >= 0 or carry:
        a = ord(s[i]) - ord('0') if i >= 0 else 0
        b = ord(t[j]) - ord('0') if j >= 0 else 0

        total = a + b + carry
        res.append(chr(total % 10 + ord('0')))
        carry = total // 10

        i -= 1
        j -= 1

    return ''.join(reversed(res))
```
