## 最小覆盖子串

题号： BM90，[LC76](https://leetcode.cn/problems/minimum-window-substring/)
描述： 给出两个字符串 s 和 t，要求在 s 中找出最短的包含 t 中所有字符的连续子串
思路： 
时间复杂度： $O(n)$
空间复杂度： $O(len(set(S+T)))$, 仅由字符集大小决定。

```python
# 写法一：字典
def minWindow(self, S: str, T: str) -> str:
    need = {}  # need[c]：t 中字符 c 需要出现多少次
    window = {}  # window[c]：当前窗口中字符 c 出现多少次
    valid = 0  # 当 valid == len(need) 代表 i,j 间的串已符合要求
    ans = ""  # 存储最小的子串
    
    for t in T:
        need[t] = need.get(t, 0) + 1

    i, j = 0, 0  # i: 滑动窗口左下标 j: 滑动窗口右下标
    while j < len(S):
	    # 向右扩展
        cj = S[j]
        if cj in need:
            window[cj] = window.get(cj, 0) + 1
            if window[cj] == need[cj]:
                valid += 1

        j += 1
        # 达到条件，左边开始收缩
        while len(need) == valid:
            ci = S[i]
            if ci in need:
                if window[ci] == need[ci]:
                    valid -= 1
                window[ci] -= 1
                # j已经+1, 故下面两行就不用+1
                if ans == "" or len(ans) > j - i:
                    ans = S[i:j]
            i += 1

    return ans

# 写法二：Counter
from collections import Counter
def minWindow(self, S: str, T: str) -> str:
	need = Counter(T)
	window = Counter()
	valid = 0
	ans = ""

	i = 0
	for j in range(len(S)):
		cj = S[j]
		if cj in need:
			window[cj] += 1
			if window[cj] == need[cj]:
				valid += 1

		while len(need) == valid:
			ci = S[i]
			if ci in need:
				if window[ci] == need[ci]:
					valid -= 1
				window[ci] -= 1
				if ans == "" or len(ans) > j - i + 1:
					ans = S[i : j + 1]
			i += 1

	return ans

# 法三： 法二的记录切片改为记录起始和长度
def minWindow(self, S: str, T: str) -> str:
	need = Counter(T)
	window = Counter()
	valid = 0
	length = len(S) + 1	# 最小覆盖子串的长度, 初始为 inf
	start = 0	# 最小覆盖子串的起始位置

	left = 0
	for right, cr in enumerate(S):
		if cr in need:
			window[cr] += 1
			if window[cr] == need[cr]:
				valid += 1

		while valid == len(need):
			cl = S[left]
			if cl in need:
				if need[cl] == window[cl]:
					valid -= 1
				window[cl] -= 1
				if length > right - left + 1:
					length = right - left + 1
					start = left

			left += 1

	if length == len(S) + 1:
		return ""

	return S[start : start + length]
```



```go
func minWindow(S string, T string) string {
	window := make(map[byte]int)
	need := make(map[byte]int)
	for _, t := range []byte(T) {
		need[t]++
	}
	valid := 0
	i, j := 0, 0
	ans := ""
	for j < len(S) {
		cj := S[j]
		if _, ok := need[cj]; ok {
			window[cj]++
			if window[cj] == need[cj] {
				valid++
			}
		}

		j++
		for ; len(need) == valid; i++ {
			ci := S[i]
			if need[ci] > 0 {
				if window[ci] == need[ci] {
					valid--
				}
				window[ci]--
			}
			if ans == "" || len(ans) > j-i {
				ans = S[i:j]
			}
		}
	}

	return ans
}
```
