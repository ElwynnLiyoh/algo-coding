## 单链表排序

题号： BM12，[LC148](https://leetcode.cn/problems/sort-list/description)

### 二分归并

时间复杂度： $O(n \log n)$
空间复杂度： $O(\log n)$，递归栈

```go
func sortInList(head *ListNode) *ListNode {
    if head == nil || head.Next == nil {
        return head
    }
    slow, fast := head, head.Next
    for fast != nil && fast.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
    }
    tmp := slow.Next
    slow.Next = nil
    # 归并排序
    left := sortInList(head)
    right := sortInList(tmp)
    return merge(left, right)
}
 
func merge(left, right *ListNode) *ListNode {
    dummy := new(ListNode)
    res := dummy
    for left != nil && right != nil {
        if left.Val < right.Val {
            res.Next = left
            left = left.Next
        } else {
            res.Next = right
            right = right.Next
        }
        res = res.Next
    }
    if left != nil {
        res.Next = left
    }
    if right != nil {
        res.Next = right
    }
    return dummy.Next
}
```

### 自底向上

时间复杂度： $O(n \log n)$
空间复杂度： $O(1)$

合并的时候可以将`tail`一并返回

**Golang版本**

```go
func sortList(head *ListNode) *ListNode {
	if head == nil || head.Next == nil {
		return head
	}

	// 计算链表长度
	length := 0
	for cur := head; cur != nil; cur = cur.Next {
		length++
	}

	dummy := &ListNode{Next: head}

	// 子链表长度：1 -> 2 -> 4 -> 8 -> ...
	for step := 1; step < length; step *= 2 {
		prev := dummy
		cur := dummy.Next

		for cur != nil {
			// 第一段
			left := cur

			// 找到第一段的末尾
			for i := 1; i < step && cur.Next != nil; i++ {
				cur = cur.Next
			}

			// 第二段
			right := cur.Next
			cur.Next = nil

			// 找到第二段的末尾，并保存下一组的起点
			cur = right
			for i := 1; i < step && cur != nil && cur.Next != nil; i++ {
				cur = cur.Next
			}
			
			// 记录下一组
			var next *ListNode
			if cur != nil {
				next = cur.Next
				cur.Next = nil
			}

			// 合并两个有序链表
			mergedHead, mergedTail := merge(left, right)

			// 接到前面的链表
			prev.Next = mergedHead
			prev = mergedTail

			// 处理下一组
			cur = next
		}
	}

	return dummy.Next
}

func merge(left, right *ListNode) (*ListNode, *ListNode) {
	dummy := &ListNode{}
	tail := dummy

	for left != nil && right != nil {
		if left.Val <= right.Val {
			tail.Next = left
			left = left.Next
		} else {
			tail.Next = right
			right = right.Next
		}
		tail = tail.Next
	}

	if left != nil {
		tail.Next = left
	} else {
		tail.Next = right
	}

	// 找到合并后的尾节点
	for tail.Next != nil {
		tail = tail.Next
	}

	return dummy.Next, tail
}
```

**Python版本**

```python
class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head or not head.next:
            return head

        cur = head
        length = 0
        while cur:
            cur = cur.next
            length += 1

        dummy = ListNode(next=head)
        step = 1

        while step < length:
            pre = dummy
            cur = dummy.next

            while cur:
                left = cur

                # 找到 left 的末尾
                for _ in range(1, step):
                    if cur.next:
                        cur = cur.next
                    else:
                        break

                # 切出 right
                right = cur.next
                cur.next = None

                # 找到 right 的末尾
                cur = right
                if cur:
                    for _ in range(1, step):
                        if cur.next:
                            cur = cur.next
                        else:
                            break

                # 保存下一组
                next_group = None
                if cur:
                    next_group = cur.next
                    cur.next = None

                # 合并
                pre.next, pre = self.merge(left, right)

                # 下一组
                cur = next_group

            step *= 2

        return dummy.next

    def merge(self, l1, l2):
        dummy = ListNode(0)
        tail = dummy

        while l1 and l2:
            if l1.val <= l2.val:
                tail.next = l1
                l1 = l1.next
            else:
                tail.next = l2
                l2 = l2.next

            tail = tail.next

        tail.next = l1 if l1 else l2

        while tail.next:
            tail = tail.next

        return dummy.next, tail
```

### 堆

最无脑简单。注意弹出堆时要将原节点后的链断开。

时间复杂度： $O(n \log n)$
空间复杂度： $O(n)$，堆的大小

```python
import heapq
ListNode.__lt__ = lambda a, b: a.val < b.val
class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head or not head.next:
            return head

        h = []

        while head:
            h.append(head)
            head = head.next

        heapq.heapify(h)

        dummy = ListNode(0)
        cur = dummy

        while h:
            node = heapq.heappop(h)

            cur.next = node
            cur = node
            cur.next = None   # 断开，因为原节点的链后可能还有节点

        return dummy.next
```
