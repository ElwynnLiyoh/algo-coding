## 单链表排序

二分归并

```go
func sortInList(head *ListNode) *ListNode {
    if head == nil || head.Next == nil {
        return head
    }
    slow, fast := head, head.Next	# fast也可以是head, 但下面的判断要改成fast.Next.Next
    for fast != nil && fast.Next != nil {
        slow = slow.Next
        fast = fast.Next.Next
    }
    tmp := slow.Next
    slow.Next = nil
    # 归并排序
    left := sortInList(head)
    right := sortInList(tmp)
    return merge(left, right)
}
 
func merge(left, right *ListNode) *ListNode {
    dummy := new(ListNode)
    res := dummy
    for left != nil && right != nil {
        if left.Val < right.Val {
            res.Next = left
            left = left.Next
        } else {
            res.Next = right
            right = right.Next
        }
        res = res.Next
    }
    if left != nil {
        res.Next = left
    }
    if right != nil {
        res.Next = right
    }
    return dummy.Next
}
```

def sortInList(self, head: ListNode) -> ListNode:
        if head == None or head.next == None:
            return head

        slow, fast = head, head
        while fast.next != None and fast.next.next != None:
            slow = slow.next
            fast = fast.next.next
        tmp = slow.next
        slow.next = None
        left = self.sortInList(head)
        right = self.sortInList(tmp)
        return self.merge(left, right)
