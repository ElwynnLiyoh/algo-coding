## 合并两个已排序的链表

```python
def Merge(self , pHead1: ListNode, pHead2: ListNode) -> ListNode:
    dummy = ListNode(-1)
    cur = dummy

    while (pHead1!=None and pHead2!=None):
        if(pHead1.val<pHead2.val):
            cur.next=pHead1
            pHead1=pHead1.next
        else:
            cur.next=pHead2
            pHead2=pHead2.next
        cur=cur.next
    if(pHead1!=None):
        cur.next=pHead1
    if(pHead2!=None):
        cur.next=pHead2

    return dummy.next
```



## 合并k个已排序的链表

分解成多个“合并两个已排序链表”

```go
func mergeKLists(lists []*ListNode) (res *ListNode) {
	for _, ls := range lists {
		res = mergeTwo(res, ls)
	}
	return
}

func mergeTwo(l1, l2 *ListNode) *ListNode {
	var dummy *ListNode = new(ListNode)
	cur := dummy
	for l1 != nil && l2 != nil {
		if l1.Val < l2.Val {
			cur.Next = l1
			l1 = l1.Next
		} else {
			cur.Next = l2
			l2 = l2.Next
		}
		cur = cur.Next
	}
	if l1 != nil {
		cur.Next = l1
	}
	if l2 != nil {
		cur.Next = l2
	}
	return dummy.Next
}

```
