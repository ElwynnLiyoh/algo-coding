## 指定区间反转

题号：BM2, [LC92](https://leetcode.cn/problems/reverse-linked-list-ii/description/)

### 头插法

**核心要义**： cur不变

```python
def reverseBetween(self , head: ListNode, m: int, n: int) -> ListNode:
    dummy=ListNode(-1)
    dummy.next=head
    pre=dummy
    # 走到待返转区间前
    for _ in range(m-1):
        pre=pre.next
    
    cur=pre.next    # cur相当于待返转区间的头节点
    for _ in range(n-m):
        tmp=cur.next
        cur.next=tmp.next
        tmp.next=pre.next  # 注意不能是cur, 因为cur始终不变，将被挤到区间尾
        pre.next=tmp
    # 不能是head, 因为头节点可能在区间
    return dummy.next   
```



## K组反转

记录好待反转区间的前一个、最后一个以及

题号： BM3, [LC25](https://leetcode.cn/problems/reverse-nodes-in-k-group/description/)


```python
class Solution:
    # 写法一
    def reverseKGroup(self, head: ListNode | None, k: int) -> ListNode | None:
        dummy = ListNode(-1, head)
        pre = dummy # 记录待反转区间的前一个节点
        end = dummy # 记录待反转区间的最后一个节点
        while end.next: # end或True都可以
            for _ in range(k):
                if end.next:
                    end = end.next
                else:
                    # break # 如果不足k个也反转，只需改成break即可
                    return dummy.next

            start = pre.next    # 待反转区间的第一个节点
            ngroup = end.next     # 下一组的第一个节点
            end.next = None     # 断开，防止后面的也反转

            pre.next = self.reverse(start) # 反转并连接

            pre = start         # 反转后，start到了当前区间尾，即下一个区间的前一个节点
            end = start         # 就和初始化pre和end为dummy一样
            start.next = ngroup   # 连接到下一组，否则pre和start找不到下一个节点

        return dummy.next

    # 写法二（更简洁）
    def reverseKGroup(self, head: ListNode | None, k: int) -> ListNode | None:
        dummy = ListNode(-1, head)
        pre = dummy # 记录待反转区间的前一个节点
        while True: # 只能True
            end = pre # 记录待反转区间的最后一个节点
            for _ in range(k):
                if end.next:
                    end = end.next
                else:
                    # break # 如果不足k个也反转，只需改成break即可
                    return dummy.next

            start = pre.next    # 待反转区间的第一个节点
            ngroup = end.next     # 下一个待反转区间的第一个节点
            end.next = None     # 断开，防止后面的也反转

            pre.next = self.reverse(start) # 反转并连接

            # 如果要求不足k个也反转，则需要这个判断
            # if end.next == None:
            #     break

            pre = start         # 反转后，start到了当前区间尾，即下一个区间的前一个节点
            pre.next = ngroup   # 连接到下一组，否则pre和start找不到下一个节点

        return dummy.next

    def reverse(self, head):
        pre = None
        while head:
            next = head.next
            head.next = pre
            pre = head
            head = next
        return pre

```



### 尾插法

```python
class Solution:
    def reverseKGroup(self, head: ListNode, k: int) -> ListNode:
        dummy = ListNode(-1)
        dummy.next = head
        prev = dummy
        while True:
            # 找到当前组的第 k 个节点
            end = prev
            for _ in range(k):
                end = end.next
                if end is None:
                    return dummy.next

            # 保存当前组的头和下一组的头
            start = prev.next
            ngroup = end.next

            # 反转 [start, end]
            last = ngroup
            cur = start
            while cur != ngroup:
                tmp = cur.next
                cur.next = last
                last = cur
                cur = tmp
            # 连接反转后的链表
            prev.next = end
            
            # start 已经变成当前组的尾节点
            prev = start
```
