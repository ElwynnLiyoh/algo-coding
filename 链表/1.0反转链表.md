## 反转链表

### 直接反转

直接反转不需要dummy

```python
def ReverseList(self , head: ListNode) -> ListNode:
        pre = None
        while head != None:
            next = head.next  # 先记住下一个位置，防止断开
            head.next = pre  # 调转方向
            # 往前移动
            pre = head
            head = next
        return pre
```



### 头插法

需要dummy

**核心要义**： head不变

```python
def ReverseList(self, head: ListNode) -> ListNode:
        dummy = ListNode(-1)
        dummy.next = head
        while head != None and head.next != None:
            tmp = head.next
            head.next = tmp.next
            tmp.next = dummy.next
            dummy.next = tmp

        return dummy.next
```



## 指定区间反转

### 头插法

**核心要义**： cur不变

```python
def reverseBetween(self , head: ListNode, m: int, n: int) -> ListNode:
        dummy=ListNode(-1)
        dummy.next=head
        pre=dummy
        for _ in range(m-1):
            pre=pre.next
        
        cur=pre.next
        for _ in range(n-m):
            tmp=cur.next
            cur.next=tmp.next
            tmp.next=pre.next  # 注意不能是cur, 因为cur始终不变，第二个循环cur!=pre.next
            pre.next=tmp
		# 不能是head, 因为头节点可能在区间
        return dummy.next   
```



## K组反转

### 头插法

```python
class Solution:
    def reverseKGroup(self, head: ListNode, k: int) -> ListNode:
        dummy = ListNode(-1)
        dummy.next = head

        prev = dummy

        while True:
            # 找到当前组的第 k 个节点
            end = prev
            for _ in range(k):
                end = end.next
                if end is None:
                    return dummy.next

            # 保存当前组的头和下一组的头
            start = prev.next
            next_group = end.next

            # 反转 [start, end]
            last = next_group
            cur = start
            while cur != next_group:
                tmp = cur.next
                cur.next = last
                last = cur
                cur = tmp

            # 连接反转后的链表
            prev.next = end

            # start 已经变成当前组的尾节点
            prev = start
```
