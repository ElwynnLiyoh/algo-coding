## 链表中倒数第k个节点

```python
def FindKthToTail(self, pHead: ListNode, k: int) -> ListNode:
        latter, former = pHead, pHead
        for _ in range(k):
            if former == None:
                return None
            former = former.next

        while former != None:
            latter = latter.next
            former = former.next

        return latter
```



## 删除链表的倒数第 n 个节点

```python
def removeNthFromEnd(self, head: ListNode, n: int) -> ListNode:
        dummy = ListNode(-1)
        dummy.next = head
        latter = former = dummy
        cur = head

        for _ in range(n + 1):  # former=head的话就不用+1
            former = former.next

        while former != None:
            latter = latter.next
            cur = cur.next
            former = former.next

        latter.next = cur.next
        return dummy.next
```
