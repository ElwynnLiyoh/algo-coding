## 快速排序

[LC912](https://leetcode.cn/problems/sort-an-array/description/)

时间复杂度： $O(n \log n)$，最坏$O(n^2)$
空间复杂度： $O(\log n)$。递归开销。

### Python

手搓版（挖坑法）

```python
class Solution:
    def sortArray(self, nums):
        self.quickSort(nums, 0, len(nums) - 1)
        return nums

    def quickSort(self, nums, left, right):
        if left >= right:
            return

        p = self.partition(nums, left, right)

        self.quickSort(nums, left, p - 1)
        self.quickSort(nums, p + 1, right)

    def partition(self, nums, left, right):
        # 选择中间位置作为 pivot，避免固定选择端点导致有序数组退化。
	    mid = (left + right) // 2
        nums[mid], nums[right] = nums[right], nums[mid]
        
        pivot = nums[right] # 挖坑
        i, j = left, right

        while i < j:
	        # 坑在j则先移动i
            while i < j and nums[i] <= pivot:
                i += 1

            nums[j] = nums[i]  # 填坑并将坑转移到i
			
            # 坑在i则先移动j
            while i < j and nums[j] >= pivot:
                j -= 1

            nums[i] = nums[j]

        nums[i] = pivot
        # 返回pivot位置
        return i
```



优化

```python
from random import randint
class Solution:
    def sortArray(self, nums: list[int]) -> list[int]:
        self.quickSort(nums, 0, len(nums) - 1)
        return nums

    def quickSort(self, nums, left, right):
        while left < right:

            # 优化：已经有序
            ordered = True
            for i in range(left, right):
                if nums[i] > nums[i + 1]:
                    ordered = False
                    break
            if ordered:
                return

            p = self.partition(nums, left, right)

            # 优化：
            # 较短区间递归，较长区间交给下一轮 while，
            # 将递归栈深度控制在 O(log n)。
            if p - left < right - p:
                self.quickSort(nums, left, p - 1)
                left = p + 1
            else:
                self.quickSort(nums, p + 1, right)
                right = p - 1

    def partition(self, nums, left, right):
        # 找随机下标
        mid = randint(left, right)
        nums[mid], nums[right] = nums[right], nums[mid]

        pivot = nums[right]  # 挖坑
        i, j = left, right
        while i < j:
            # 坑在j则先移动i
            while i < j and nums[i] <= pivot:
                i += 1

            nums[j] = nums[i]  # 填坑并将坑转移到i

            while i < j and nums[j] >= pivot:
                j -= 1

            nums[i] = nums[j]

        nums[i] = pivot
        # 返回pivot位置
        return i
```
