## 6.0比较版本号

题号： BM22
时间复杂度： O(n)
空间复杂度： O(1)

```python
def compare(self, version1: str, version2: str) -> int:
        len1, len2 = len(version1), len(version2)
        i, j = 0, 0
        while i < len1 or j < len2:
            v1, v2 = 0, 0
            while i < len1 and version1[i] != ".":
              	# 天然跳过前导0
                v1 = v1 * 10 + int(version1[i]) - int("0")
                i += 1
            while j < len2 and version2[j] != ".":
                v2 = v2 * 10 + int(version2[j]) - int("0")
                j += 1
            
            if v1 > v2:
                return 1
            elif v1 < v2:
                return -1
            # 跳过 '.'
            i += 1
            j += 1
        return 0

```
