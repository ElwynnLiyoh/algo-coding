## 5.0数组中的​逆序对

题号： BM20
时间复杂度： O(n)
空间复杂度： O(nlogn)

```go
func InversePairs(nums []int) int {
	mergeSort(nums)
	return sum
}

// 归并排序
func mergeSort(arr []int) []int {
	if len(arr) <= 1 {
		return arr
	}

	mid := len(arr) >> 1
	left := mergeSort(arr[:mid])
	right := mergeSort(arr[mid:])

	return merge(left, right)
}

var sum int

// 合并两个有序数组
func merge(arr1 []int, arr2 []int) []int {
	len1, len2 := len(arr1), len(arr2)
	merged := make([]int, len1+len2)
	i, j, k := 0, 0, 0

	for i < len1 && j < len2 {
		if arr1[i] <= arr2[j] {
			// merged = append(merged, arr1[i])
			merged[k] = arr1[i]
			i++
		} else {
			// merged = append(merged, arr2[j])
			merged[k] = arr2[j]
			j++

			// 剩余 arr1[i:] 全部比 arr2[j] 大
			sum += len1 - i
		}
		k++
	}

	// merged = append(merged, arr1[i:]...)
	// merged = append(merged, arr2[j:]...)
	for ; i < len1; i++ {
		// merged = append(merged, arr1[i])
		merged[k] = arr1[i]
		k++
	}
	for ; j < len2; j++ {
		// merged = append(merged, arr2[j])
		merged[k] = arr2[j]
		k++
	}

	return merged
}
```
