## 寻找峰值
题号： BM19
时间复杂度： O(logN)
空间复杂度： O(1)

上坡一定能找到峰值，下坡可能有峰值。所以应该往上走
```python
def findPeakElement(self, nums: List[int]) -> int:
    left, right = 0, len(nums) - 1
    while left < right:
        mid = left + (right - left) // 2
        if nums[mid] > nums[mid + 1]:  # 往右是下坡，未必有峰值
            right = mid
        else:   # 往右是上坡，一定有峰值
            left = mid + 1

    return left
```



```python
def findPeakElement(self, nums: List[int]) -> int:
	left, right = - 1, len(nums) - 1
	while left + 1 < right:
		mid = left + (right - left) // 2
		if nums[mid] > nums[mid + 1]: 
			right = mid
		else: 
			left = mid

	return right
```



## 旋转数组的最小数字

题号： BM21
时间复杂度： O(logN)
空间复杂度： O(1)

```go
func minNumberInRotateArray(nums []int) int {
	left, right := 0, len(nums)-1
	for left < right {
		mid := left + (right-left)/2
		// 每次比较mid和右端点的值
		if nums[mid] > nums[right] {
			//mid、right位于两段，最小值在右侧[mid+1,right]
			left = mid + 1
		} else if nums[mid] < nums[right] {
			//mid、right位于一段，最小值在左侧[left,mid]，不排除最小值就是mid
			right = mid
		} else {
			// 两端或一段都可能，只能将 right 指针左移一位
			right--
		}
	}
	return nums[left]
}
```
