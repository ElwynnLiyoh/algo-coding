## 3.0颜色分类(荷兰国旗问题)
题号： NC212
时间复杂度： O()
空间复杂度： O()

注意左右两边对idx的处理不同

循环条件： idx <= right

```python
def sortColor(self, colors: List[int]) -> List[int]:
    left, right = 0, len(colors) - 1
    idx = left
    while idx <= right:
        if colors[idx] == 0:
            colors[idx], colors[left] = colors[left], 0
            left += 1
            idx += 1
        elif colors[idx] == 2:
            colors[idx], colors[right] = colors[right], 2
            right -= 1
        else:
            idx += 1
    return colors

```



```go
func sortColor(colors []int) []int {
    left, i, right := 0, 0, len(colors)-1

    for i <= right {
        if colors[i] == 0 {
            colors[i], colors[left] = colors[left], colors[i]
            left++
            i++
        } else if colors[i] == 2 {
            colors[i], colors[right] = colors[right], colors[i]
            right--
          	//注意这里没有i++
        } else {
            // colors[i] == 1
            i++
        }
    }

    return colors
}
```
