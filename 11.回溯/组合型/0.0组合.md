## 组合

题号：[LC77](https://leetcode.cn/problems/combinations/description/)
描述：给定两个整数 `n` 和 `k`，返回范围 `[1, n]` 中所有可能的 `k` 个数的组合。
思路：
时间复杂度： $O()$
空间复杂度： $O()$

```python
class Solution:
    # 法一：倒着走，枚举
    def combine(self, n: int, k: int) -> list[list[int]]:
        ans = []
        path = []

        def dfs(i):
            # 还剩几个数（还需要几个数）
            d = k - len(path)
            # 剪枝
            if i < d:
                return

            if len(path) == k: # d==0也可
                ans.append(path.copy())
                return
            
            # 倒着走
            # 剪枝也可以是range(i, n-d, -1)
            for j in range(i, 0, -1): # 范围最小是1
                path.append(j)
                dfs(j - 1)
                path.pop()

        dfs(n) # 倒着所以是n
        return ans

	# 法二：选或不选
    def combine(self, n: int, k: int) -> List[List[int]]:
        ans = []
        path = []

        # 选或不选：讨论 i 是否加入 path
        def dfs(i: int) -> None:
            d = k - len(path)  # 还要选 d 个数
            if d == 0:  # 选好了
                ans.append(path.copy())
                return

            # 不选 i。因为不选，所以不能等于，而要大于
            if i > d:
                dfs(i - 1)

            # 选 i
            path.append(i)
            dfs(i - 1)
            path.pop()  # 恢复现场

        dfs(n)  # 从 i=n 开始倒着枚举
        return ans
    
    # 法三：正着走
    def combine(self, n: int, k: int) -> list[list[int]]:
        ans = []
        path = []

        def dfs(i):
            # 还剩几个数
            d = k - len(path)
            if n - i + 1 < d: # +1是因为范围是从1开始
                return

            # if len(path) == k:
            if d == 0:
                ans.append(path.copy())
                return

            # 因为可以取到n
            for j in range(i, n + 1):
                path.append(j)
                dfs(j + 1)
                path.pop()

        dfs(1)
        return ans

```
