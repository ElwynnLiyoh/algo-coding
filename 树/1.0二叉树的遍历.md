## 二叉树的遍历

| 遍历方式     | 适用     | 数据结构 |
| ------------ | -------- | -------- |
| 深度优先 DFS | 前中后序 | 栈       |
| 广度优先 BFS | 层序     | 队列     |

序都是指**根**的序。



### 前序遍历

题号： BM23
描述： 
思路： 
时间复杂度： $O(n)$
空间复杂度： $O(n)$，最坏情况下二叉树化为链表，递归栈深度为n

递归

```python
class Solution:
    def preorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.preorder(res, root)
        return res

    def preorder(self, res: List[int], root: TreeNode):
        if root == None:
            return
        res.append(root.val)
        self.preorder(res, root.left)
        self.preorder(res, root.right)
```

栈

先访问的后进栈

```python
class Solution:
    def preorderTraversal(self , root: TreeNode) -> List[int]:
        res = []
        if not root:
            return res
        # 辅助栈
        s = [] 
        # 根节点先进栈
        s.append(root) 
        # 直到栈中没有节点
        while s: 
            # 每次栈顶就是访问的元素
            # node = s[-1] 
            node = s.pop()
            res.append(node.val)
            # 如果右边还有右子节点进栈
            if node.right: 
                s.append(node.right)
            # 如果左边还有左子节点进栈
            if node.left: 
                s.append(node.left)
        return res

```



### 中序遍历

题号： BM24

递归

```python
class Solution:
    def inorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.inorder(res, root)
        return res

    def inorder(self, res: List[int], root: TreeNode):
        if not root:
            return None
        self.inorder(res, root.left)
        res.append(root.val)
        self.inorder(res, root.right)
```

栈

```python
class Solution:
    def inorderTraversal(self , root: TreeNode) -> List[int]:
        # s是辅助栈
        res, s = [], [] 
        # 当树节点不为空或栈中有节点时
        while root or s: 
            # 每次找到最左节点
            while root: 
                s.append(root)
                root = root.left
            # 访问该节点
            node = s.pop() 
            res.append(node.val)
            # 进入右节点
            root = node.right 
        return res

```



### 后序遍历

题号： BM25

递归

```python
class Solution:
    def postorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.postorder(res, root)
        return res

    def postorder(self, res: List[int], root: TreeNode):
        if not root:
            return None

        self.postorder(res, root.left)
        self.postorder(res, root.right)
        res.append(root.val)
```

栈

```python
class Solution:
    def postorderTraversal(self , root: TreeNode) -> List[int]:
        res, s = [], [] 
        pre = None
        while root or s:
            # 每次先找到最左边的节点
            while root: 
                s.append(root)
                root = root.left
            # 弹出栈顶
            node = s.pop()
            # 如果该元素的右边没有或是已经访问过
            if not node.right or node.right is pre: 
                # 访问中间的节点
                res.append(node.val) 
                # 且记录为访问过了
                pre = node 
            else:
                # 该节点入栈
                s.append(node) 
                # 先访问右边
                root = node.right 
        return res

```



### 层序遍历

题号：BM26，LC102

描述： 给定一个二叉树，返回该二叉树层序遍历的结果，（从左到右，一层一层地遍历）

思路：

1. 初始化一个辅助队列
2. 根结点入队
3. 「循环」若队列非空，则队头结点出队，访问该结点，并将其左右孩子插入队尾（如有）

队列

```python
import queue
class Solution:
    def levelOrder(self , root: TreeNode) -> List[List[int]]:
        res = []
        if not root:
            return res
        
        q = queue.Queue()
        q.put(root)
        while not q.empty():
            row = []    # 记录一层
            # python里range()只计算一次，故q.qsize()会变化也没关系
            for _ in range(q.qsize()):  
                node = q.get()
                row.append(node.val)
                if node.left:
                    q.put(node.left)
                if node.right:
                    q.put(node.right)
                    
            res.append(row)

        return res
```

go版本

```go
func levelOrder(root *TreeNode) [][]int {
    res := [][]int{}
    if root == nil {
      	return res
    }

    // 先放入根节点
    queue := []*TreeNode{root}
    for len(queue) > 0 {
        level := []int{} // 记录一层
        size := len(queue)
        for i := 0; i < size; i++ {
            // 队头
            node := queue[0]
            queue = queue[1:]

            level = append(level, node.Val)
            if node.Left != nil {
              	queue = append(queue, node.Left)
            }
            if node.Right != nil {
              	queue = append(queue, node.Right)
            }
        }
        res = append(res, level)
    }

    return res
}
```


递归

```go
var res = [][]int{}

func levelOrder(root *TreeNode) [][]int {
	traverse(root, 1)
	return res
}

func traverse(root *TreeNode, level int) {
	if root != nil {
		// 新的一层
		if len(res) < level {
			res = append(res, []int{})
		}
		// 元素添加入这一层
		res[level-1] = append(res[level-1], root.Val)
		// 递归时层数+1
		traverse(root.Left, level+1)
		traverse(root.Right, level+1)
	}
}
```
