## 最小覆盖子串

题号： BM90
描述： 给出两个字符串 s 和 t，要求在 s 中找出最短的包含 t 中所有字符的连续子串
思路： 
时间复杂度： $O(n)$
空间复杂度： $O(n)$

```python
def minWindow(self, S: str, T: str) -> str:
    need = {}  # need[c]：t 中字符 c 需要出现多少次
    window = {}  # window[c]：当前窗口中字符 c 出现多少次
    valid = 0  # 当 valid == len(need) 代表 i,j 间的h串已符合要求
    ans = ""  # 存储最小的子串
    for t in T:
        need[t] = need.get(t, 0) + 1

    i, j = 0, 0  # i: 滑动窗口左下标 j: 滑动窗口右下标
    while j < len(S):
        cj = S[j]
        if cj in need:
            window[cj] = window.get(cj, 0) + 1
            if window[cj] == need[cj]:
                valid += 1

        j += 1
        while len(need) == valid:
            ci = S[i]
            if ci in need:
                if window[ci] == need[ci]:
                    valid -= 1
                window[ci] -= 1
                # j已经+1, 故下面两行就不用+1
                if ans == "" or len(ans) > j - i:
                    ans = S[i:j]
            i += 1

    return ans
```



```go
func minWindow(S string, T string) string {
	window := make(map[byte]int)
	need := make(map[byte]int)
	for _, t := range []byte(T) {
		need[t]++
	}
	valid := 0
	i, j := 0, 0
	ans := ""
	for j < len(S) {
		cj := S[j]
		if _, ok := need[cj]; ok {
			window[cj]++
			if window[cj] == need[cj] {
				valid++
			}
		}

		j++
		for ; len(need) == valid; i++ {
			ci := S[i]
			if need[ci] > 0 {
				if window[ci] == need[ci] {
					valid--
				}
				window[ci]--
			}
			if ans == "" || len(ans) > j-i {
				ans = S[i:j]
			}
		}
	}

	return ans
}
```
