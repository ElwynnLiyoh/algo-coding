## 螺旋矩阵

题号： BM98，LC54
描述： 给定一个m x n大小的矩阵（m行，n列），按顺时针螺旋的顺序返回矩阵中的所有元素。
思路： 需要维护上下左右的边界
时间复杂度： $O(mn)$
空间复杂度： $O(mn)$

全部左闭右开，上闭下开
```python
def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
    if not matrix:
        return []
    top, right, bottom, left = 0, len(matrix[0]), len(matrix), 0
    visit = []
    while True:
        for i in range(left, right):
            visit.append(matrix[top][i])
        top += 1
        if top == bottom:
            break

        for i in range(top, bottom):
            visit.append(matrix[i][right - 1])
        right -= 1
        if left == right:
            break

        for i in range(right - 1, left - 1, -1):
            visit.append(matrix[bottom - 1][i])
        bottom -= 1
        if top == bottom:
            break

        for i in range(bottom - 1, top - 1, -1):
            visit.append(matrix[i][left])
        left += 1
        if left == right:
            break

    return visit
```


go版本
```go
func spiralOrder(matrix [][]int) []int {
    if len(matrix) == 0 {
        return []int{}
    }
    top, right, bottom, left := 0, len(matrix[0]), len(matrix), 0
    visit := []int{}
    for {
        for i := left; i < right; i++ {
            visit = append(visit, matrix[top][i])
        }
        top++
        if top == bottom {
            break
        }

        for i := top; i < bottom; i++ {
            visit = append(visit, matrix[i][right-1])
        }
        right--
        if left == right {
            break
        }

        for i := right - 1; i >= left; i-- {
            visit = append(visit, matrix[bottom-1][i])
        }
        bottom--
        if top == bottom {
            break
        }

        for i := bottom - 1; i >= top; i-- {
            visit = append(visit, matrix[i][left])
        }
        left++
        if left == right {
            break
        }
    }

    return visit
}

func spiralOrder(matrix [][]int) []int {
	if len(matrix) == 0 {
		return []int{}
	}
	top, right, bottom, left := 0, len(matrix[0]), len(matrix), 0
	visit := []int{}
	// 换种判断方式，避免太多break
	for left < right && top < bottom {
		for i := left; i < right; i++ {
			visit = append(visit, matrix[top][i])
		}
		top++

		for i := top; i < bottom; i++ {
			visit = append(visit, matrix[i][right-1])
		}
		right--
		
		// 先判断，否则会边界溢出
		if top < bottom {
			for i := right - 1; i >= left; i-- {
				visit = append(visit, matrix[bottom-1][i])
			}
		}
		bottom--

		if left < right {
			for i := bottom - 1; i >= top; i-- {
				visit = append(visit, matrix[i][left])
			}
		}
		left++
	}

	return visit
}
```
