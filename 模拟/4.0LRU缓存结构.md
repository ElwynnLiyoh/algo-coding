## LRU缓存结构

题号： BM100
描述： LRU Cache（Least Recently Used, 最近最少使用缓存）
思路： 哈希表 + 双向链表，使用虚拟头尾结点插入和尾删除都是O(1)
时间复杂度： $O(1)$
空间复杂度： $O(n)$

简化版：
CPython 的 `dict` 实际上已经保持插入顺序，但这属于 **CPython 的实现细节**，语言规范当时没有保证。
- 字典开头 → 最久未使用
- 字典末尾 → 最近使用
```python
class Solution:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1

        # 删除后重新插入，使其成为最近使用
        value = self.cache.pop(key)
        self.cache[key] = value

        return value

    def set(self, key: int, value: int):
        if key in self.cache:
            # 删除旧位置
            del self.cache[key]

        # 插入到最后，表示最近使用
        self.cache[key] = value

        # 超过容量，删除最久未使用的第一个元素
        if len(self.cache) > self.capacity:
            oldest_key = next(iter(self.cache))
            del self.cache[oldest_key]
```

```python
from collections import OrderedDict

class Solution:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1

        # 移动到最后，表示最近使用
        self.cache.move_to_end(key)

        return self.cache[key]

    def set(self, key: int, value: int):
        if key in self.cache:
            # 修改值
            self.cache[key] = value

            # 变为最近使用
            self.cache.move_to_end(key)
        else:
            # 新元素默认插入末尾
            self.cache[key] = value

            # 超过容量，删除最久未使用的元素
            if len(self.cache) > self.capacity:
                self.cache.popitem(last=False)
                
    # 更简洁的set
    def set(self, key: int, value: int):
        if key in self.cache:
            self.cache.move_to_end(key)

        self.cache[key] = value

        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)
```

手搓版：
```python
class Node:
    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class Solution:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}

        # 虚拟头尾节点
        self.head = Node()
        self.tail = Node()

        self.head.next = self.tail
        self.tail.prev = self.head

    # 删除一个节点
    def remove(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    # 将节点插入头部
    def add_to_head(self, node):
        node.prev = self.head
        node.next = self.head.next

        self.head.next.prev = node
        self.head.next = node

    # 将节点变成最近使用
    def move_to_head(self, node):
        self.remove(node)
        self.add_to_head(node)

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1

        node = self.cache[key]

        # 被访问，更新为最近使用
        self.move_to_head(node)

        return node.value

    def set(self, key: int, value: int) -> None:
        if key in self.cache:
            # 修改 value
            node = self.cache[key]
            node.value = value

            # 更新使用顺序
            self.move_to_head(node)
        else:
            node = Node(key, value)
            # 加入哈希表
            self.cache[key] = node
            # 加入链表头部
            self.add_to_head(node)
            # 超过容量
            if len(self.cache) > self.capacity:
                # 最久未使用的节点
                remove_node = self.tail.prev
                # 从链表删除
                self.remove(remove_node)
                # 从哈希表删除
                del self.cache[remove_node.key]
```



Go手搓版本：

```go
type Node struct {
    key   int
    value int
    prev  *Node
    next  *Node
}

type LRUCache struct {
    capacity int
    cache    map[int]*Node
    head     *Node
    tail     *Node
}

func Constructor(capacity int) LRUCache {
    head := &Node{}
    tail := &Node{}

    head.next = tail
    tail.prev = head

    return LRUCache{
        capacity: capacity,
        cache:    make(map[int]*Node),
        head:     head,
        tail:     tail,
    }
}

// 删除节点
func (l *LRUCache) remove(node *Node) {
    node.prev.next = node.next
    node.next.prev = node.prev
}

// 把节点放到头部
func (l *LRUCache) addToHead(node *Node) {
    node.next = l.head.next
    node.prev = l.head

    l.head.next.prev = node
    l.head.next = node
}

func (l *LRUCache) Get(key int) int {
    node, ok := l.cache[key]
    if !ok {
        return -1
    }

    // 使用过，移动到头部
    l.remove(node)
    l.addToHead(node)

    return node.value
}

func (l *LRUCache) Put(key int, value int) {
    // 已经存在
    if node, ok := l.cache[key]; ok {
        node.value = value

        // 更新为最近使用
        l.remove(node)
        l.addToHead(node)
        return
    }

    // 新节点
    node := &Node{
        key:   key,
        value: value,
    }

    l.cache[key] = node
    l.addToHead(node)

    // 超过容量
    if len(l.cache) > l.capacity {
        // tail.prev 就是最久未使用的节点
        oldest := l.tail.prev

        l.remove(oldest)
        delete(l.cache, oldest.key)
    }
}
```



### 手搓总结

- 双链表
  - Node结构体（键值、前后指针）
  - 表头：最近使用过    表尾：最久未使用
- LRU缓存结构
  - 容量
  - Hash Map
  - 头尾结点
  - 配套方法
    - 删除
    - 移到表头
