## 链表相加

题号： BM11, [LC2](https://leetcode.cn/problems/add-two-numbers/description/)

有dummy，最后还要注意是否有进位。

Python（LC版）

```python
def addTwoNumbers(self, l1: ListNode, l2: ListNode) -> ListNode | None:
    dummy = ListNode(-1)
    cur = dummy
    carry = 0

    while l1 or l2:
        node = ListNode(carry)
        if l1:
            node.val += l1.val
            l1 = l1.next
        if l2:
            node.val += l2.val
            l2 = l2.next
        carry = node.val // 10
        node.val %= 10
        cur.next = node
        cur = cur.next

    if carry == 1:
        cur.next = ListNode(1)

    return dummy.next
```



Go（NC版）


```go
func reverseList(head *ListNode) *ListNode {
    pre := (*ListNode)(nil)
    for head != nil {
        nx := head.Next
        head.Next = pre
        pre = head
        head = nx
    }
    return pre
}
 
func addInList(head1 *ListNode, head2 *ListNode) *ListNode {
    rev1 := reverseList(head1)
    rev2 := reverseList(head2)
    carryBit := 0
    dummy := new(ListNode)
    cur := dummy
    for rev1 != nil || rev2 != nil {
        newNode := ListNode{Val: carryBit}
        if rev1 != nil {
            newNode.Val += rev1.Val
            rev1 = rev1.Next
        }
        if rev2 != nil {
            newNode.Val += rev2.Val
            rev2 = rev2.Next
        }
        carryBit = newNode.Val / 10
        newNode.Val %= 10
        cur.Next  = &newNode
        cur = cur.Next
    }
    if carryBit == 1 {
        cur.Next = &ListNode{Val: 1}
    }
    return reverseList(dummy.Next)
}
```
