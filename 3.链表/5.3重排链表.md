## 重排链表

题号： [LC143](https://leetcode.cn/problems/reorder-list/)
描述：
![示例图](https://pic.leetcode.cn/1626420320-YUiulT-image.png)

### 迭代

思路： 走到中点，后半反转，再穿插。
时间复杂度： $O(n)$
空间复杂度： $O(1)$

```python
class Solution:
    def reorderList(self, head: Optional[ListNode]) -> None:
        mid = self.middleNode(head)

        h1 = head
        h2 = self.reverse(mid)
        # 因为中点前没断开，所以当数量为偶数时，h1链长>h2链长
        while h2.next:  # 等价于 h2 != mid
            # 保存下一个位置
            nx1 = h1.next
            nx2 = h2.next
            # 连接
            h1.next = h2
            h2.next = nx1
            # 更新指针
            h1 = nx1
            h2 = nx2

    def middleNode(self, head):
        slow, fast = head, head
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
        return slow

    def reverse(self, head):
        pre = None
        while head:
            nx = head.next
            head.next = pre
            pre = head
            head = nx
        return pre
```

更标准的解法，返回中点的前一个节点

```python
class Solution:
    def reorderList(self, head: Optional[ListNode]) -> None:
        #中点的前一个节点
        mid = self.middleNode(head)

        h1 = head
        h2 = self.reverse(mid.next)  # 保证h1不短于h2
        mid.next = None  # 断开中点前的链表
        
        # 只要h2还有节点，就往h1插入
        while h2:
            # 保存下一个位置
            nx1 = h1.next
            nx2 = h2.next
            # 连接
            h1.next = h2
            h2.next = nx1
            # 更新指针
            h1 = nx1
            h2 = nx2

    def middleNode(self, head):
        # fast 初始化为 head.next
        slow, fast = head, head.next
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
        # slow 指向中点的前一个节点
        return slow

    def reverse(self, head):
        pre = None
        while head:
            nx = head.next
            head.next = pre
            pre = head
            head = nx
        return pre
```

### 递归

时间复杂度： $O(n^2)$，每次都要找到尾节点
空间复杂度： $O(n)$，递归栈

```python
class Solution:
    def reorderList(self, head: ListNode) -> None:
        if not head or not head.next or not head.next.next:
            return

        # 找到尾节点及其前一个节点
        prev = head
        tail = head.next

        while tail.next:
            prev = tail
            tail = tail.next

        # 把尾节点从原位置摘下来
        prev.next = None

        # 把尾节点插入到 head 后面
        nxt = head.next
        head.next = tail
        tail.next = nxt

        # 递归处理剩余部分
        self.reorderList(nxt)
```
