## 二叉树的层序遍历

题号：BM26，[LC102](https://leetcode.cn/problems/binary-tree-level-order-traversal)
描述： 给定一个二叉树，返回该二叉树层序遍历的结果，（从左到右，一层一层地遍历）

### 队列（BFS）

1. 初始化一个辅助队列
2. 根结点入队
3. 「循环」若队列非空，则队头结点出队，访问该结点，并将其左右孩子插入队尾（如有）

时间复杂度： $O(n)$
空间复杂度： $O(n)$，队列最多在最下面一层，为`n//2+1`。

队列（标准解法）

```python
import queue
class Solution:
    def levelOrder(self , root: TreeNode) -> List[List[int]]:
        res = []
        if not root:
            return res
        
        q = queue.Queue()
        q.put(root)
        while not q.empty():
            row = []    # 记录一层
            # python里range()只计算一次，故q.qsize()会变化也没关系
            for _ in range(q.qsize()):  
                node = q.get()
                row.append(node.val)
                if node.left:
                    q.put(node.left)
                if node.right:
                    q.put(node.right)
                    
            res.append(row)

        return res
```

使用**动态数组**代替队列

```python
class Solution:
    def levelOrder(self, root: TreeNode | None) -> list[list[int]]:
        res = []
        if root is None:
            return res

        cur = [root]

        while cur:
            nxt = []
            row = []
            for node in cur:
                row.append(node.val)
                if node.left:
                    nxt.append(node.left)
                if node.right:
                    nxt.append(node.right)
            res.append(row)
            cur = nxt

        return res
```


go版本

```go
func levelOrder(root *TreeNode) [][]int {
    res := [][]int{}
    if root == nil {
      	return res
    }

    // 先放入根节点
    queue := []*TreeNode{root}
    for len(queue) > 0 {
        level := []int{} // 记录一层
        size := len(queue)
        for i := 0; i < size; i++ {
            // 队头
            node := queue[0]
            queue = queue[1:]

            level = append(level, node.Val)
            if node.Left != nil {
              	queue = append(queue, node.Left)
            }
            if node.Right != nil {
              	queue = append(queue, node.Right)
            }
        }
        res = append(res, level)
    }

    return res
}
```


### 递归（DFS）

时间复杂度： $O(n)$
空间复杂度： $O(n)$

```python
class Solution:
    def levelOrder(self, root: TreeNode | None) -> list[list[int]]:
        res = []

        def dfs(node, level):
            if not node:
                return
            # 第一次访问这一层时，创建一个新的列表
            if len(res) == level:
                res.append([])

            res[level].append(node.val)

            # 层数+1, 递归处理左右子树
            level += 1
            dfs(node.left, level)
            dfs(node.right, level)

        dfs(root, 0)
        return res
```

**go**

```go
var res = [][]int{}

func levelOrder(root *TreeNode) [][]int {
	traverse(root, 1)
	return res
}

func traverse(root *TreeNode, level int) {
	if root != nil {
		// 新的一层
		if len(res) < level {
			res = append(res, []int{})
		}
		// 元素添加入这一层
		res[level-1] = append(res[level-1], root.Val)
		// 递归时层数+1
		traverse(root.Left, level+1)
		traverse(root.Right, level+1)
	}
}
```
