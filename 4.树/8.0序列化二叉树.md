## 序列二叉树

题号： BM39
描述： 请实现两个函数，分别用来序列化和反序列化二叉树，不对序列化之后的字符串进行约束，但要求能够根据序列化之后的字符串重新构造出一棵与原二叉树相同的树。

除了中序，都可以序列化和反序列化
### 层序序列化

时间复杂度： $O(n)$
空间复杂度： $O(n)$

```python
import queue
class Solution:
    def Serialize(self, root):
        if not root:
            return '#'

        q = queue.Queue()
        q.put(root)

        res = []

        while not q.empty():
            node = q.get()

            if node:
                res.append(str(node.val))
                q.put(node.left)
                q.put(node.right)
            else:
                res.append('#')

        return ','.join(res)

    def Deserialize(self, s):
        if not s or s == '#':
            return None

        data = s.split(',')
        root = TreeNode(int(data[0]))

        q = queue.Queue()
        q.put(root)

        index = 1
        while not q.empty() :
            node = q.get()

            # 左孩子
            if index < len(data) and data[index] != '#':
                node.left = TreeNode(int(data[index]))
                q.put(node.left)
            index += 1

            # 右孩子
            if index < len(data) and data[index] != '#':
                node.right = TreeNode(int(data[index]))
                q.put(node.right)
            index += 1

        return root
```


### 前序序列化

```python
class Solution:
    def Serialize(self, root):
        if not root:
            return "#"
  
        return (
            str(root.val)
            + ","
            + self.Serialize(root.left)
            + ","
            + self.Serialize(root.right)
        )
  
    def Deserialize(self, s: str) -> TreeNode:
  
        data = s.split(",")
        index = 0
  
        def dfs():
            nonlocal index
  
            if data[index] == "#":
                index += 1
                return None
  
            node = TreeNode(int(data[index]))
            index += 1
  
            node.left = dfs()
            node.right = dfs()
  
            return node
  
        return dfs()
```


### 后序序列化

后序需要和前序反着来
```python
def Serialize(self, root):
    if not root:
        return '#'

    return (
        self.Serialize(root.left) + ','
        + self.Serialize(root.right) + ','
        + str(root.val)
    )

def Deserialize(self, s):
	# 从后往前
    index = len(s) - 1

    def dfs():
        nonlocal index

        if s[index] == '#':
            index -= 1
            return None

        node = TreeNode(int(s[index]))
        index -= 1
		# 先右后左
        node.right = dfs()
        node.left = dfs()

        return node

    return dfs()
```
