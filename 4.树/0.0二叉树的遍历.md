## 二叉树的遍历

| 遍历方式     | 适用     | 数据结构 |
| ------------ | -------- | -------- |
| 深度优先 DFS | 前中后序 | 栈       |
| 广度优先 BFS | 层序     | 队列     |

序都是指**根**的序。



### 前序遍历

题号： BM23，[LC144](https://leetcode.cn/problems/binary-tree-preorder-traversal/description/)

时间复杂度： $O(n)$
空间复杂度： $O(n)$，最坏情况下二叉树化为链表，递归栈深度为n

**递归**

```python
class Solution:
    def preorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.preorder(res, root)
        return res

    def preorder(self, res: List[int], root: TreeNode):
        if root == None:
            return
        res.append(root.val)
        self.preorder(res, root.left)
        self.preorder(res, root.right)
```

**栈**

先访问的后进栈，故应右孩子先进栈。

```python
class Solution:
    def preorderTraversal(self , root: TreeNode) -> List[int]:
        res = []
        if not root:
            return res
        # 辅助栈
        s = [] 
        # 根节点先进栈
        s.append(root) 
        # 直到栈中没有节点
        while s: 
            # 每次栈顶就是访问的元素
            # node = s[-1] 
            node = s.pop()
            res.append(node.val)
            # 如果右边还有右子节点进栈
            if node.right: 
                s.append(node.right)
            # 如果左边还有左子节点进栈
            if node.left: 
                s.append(node.left)
        return res

```



### 中序遍历

题号： BM24，[LC94](https://leetcode.cn/problems/binary-tree-inorder-traversal/)

**递归**

```python
class Solution:
    def inorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.inorder(res, root)
        return res

    def inorder(self, res: List[int], root: TreeNode):
        if not root:
            return None
        self.inorder(res, root.left)
        res.append(root.val)
        self.inorder(res, root.right)
```

**栈**

```python
class Solution:
    def inorderTraversal(self , root: TreeNode) -> List[int]:
        # s是辅助栈
        res, s = [], [] 
        # 当树节点不为空或栈中有节点时
        while root or s: 
            # 每次找到最左节点
            while root: 
                s.append(root)
                root = root.left
            # 访问该节点
            node = s.pop() 
            res.append(node.val)
            # 进入右节点
            root = node.right 
        return res

```

### 后序遍历

题号： BM25，[LC145](https://leetcode.cn/problems/binary-tree-postorder-traversal/description/)

**递归**

```python
class Solution:
    def postorderTraversal(self, root: TreeNode) -> List[int]:
        res = []
        self.postorder(res, root)
        return res

    def postorder(self, res: List[int], root: TreeNode):
        if not root:
            return None

        self.postorder(res, root.left)
        self.postorder(res, root.right)
        res.append(root.val)
```

**栈**

```python
class Solution:
    def postorderTraversal(self , root: TreeNode) -> List[int]:
        res, s = [], [] 
        pre = None
        while root or s:
            # 每次先找到最左边的节点
            while root: 
                s.append(root)
                root = root.left
            # 弹出栈顶
            node = s.pop()
            # 如果该元素的右边没有或是已经访问过
            if not node.right or node.right is pre: 
                # 访问中间的节点
                res.append(node.val) 
                # 且记录为访问过了
                pre = node 
            else:
                # 该节点入栈
                s.append(node) 
                # 先访问右边
                root = node.right 
        return res
```
