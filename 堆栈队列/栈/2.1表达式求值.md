## 表达式求值

题号： BM49
描述： 请写一个整数计算器，支持加减乘三种运算和括号。
思路： 中缀表达式和后缀表达式计算的结合

### 双栈

操作符栈 + 操作数栈

时间复杂度： $O(n)$
空间复杂度： $O(n)$

```python
class Solution:
    def solve(self, s: str) -> int:
        nums = []
        ops = []

        def calc():
            b = nums.pop()
            a = nums.pop()
            op = ops.pop()

            if op == '+':
                nums.append(a + b)
            elif op == '-':
                nums.append(a - b)
            elif op == '*':
                nums.append(a * b)
            else:
                nums.append(int(a / b))

        i = 0
        while i < len(s):
            c = s[i]

            # 解析数字
            if c.isdigit():
                num = 0
                while i < len(s) and s[i].isdigit():
                    num = num * 10 + int(s[i])
                    i += 1
                nums.append(num)
                continue

            # 左括号
            if c == '(':
                ops.append(c)

            # 右括号：计算到左括号
            elif c == ')':
                while ops[-1] != '(':
                    calc()
                ops.pop() # 弹出左括号

            # 运算符
            else:
                while (ops and
                       ops[-1] != '(' and
                       self.priority(ops[-1]) >= self.priority(c)):
                    calc()
                ops.append(c)

            i += 1

        # 处理剩余运算
        while ops:
            calc()

        return nums[-1]

    def priority(self, op):
        if op in '*/':
            return 2
        return 1
```



### 递归+栈

遇到括号即递归

```python
class Solution:
    def solve(self, s: str) -> int:
        def dfs(i):
            stack = []
            num = 0
            op = '+'

            while i < len(s):
                c = s[i]

                if c.isdigit():
                    num = num * 10 + int(c)

                if c == '(':
                    num, i = dfs(i + 1)

                if c in '+-*/)' or i == len(s) - 1:
                    if op == '+':
                        stack.append(num)
                    elif op == '-':
                        stack.append(-num)
                    elif op == '*':
                        stack.append(stack.pop() * num)
                    elif op == '/':
                        stack.append(int(stack.pop() / num))

                    num = 0
                    op = c

                if c == ')':
                    return sum(stack), i

                i += 1

            return sum(stack)

        return dfs(0)
```
